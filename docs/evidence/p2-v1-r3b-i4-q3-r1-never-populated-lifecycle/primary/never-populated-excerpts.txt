  250          if (!pidref_is_set(pidref))
  251                  return -ESRCH;
  252
  253          if (pidref->pid == 1)
  254                  return 1; /* PID 1 can never go away, hence never be recycled to a different process → return 1 */
  255
  256          if (pidref->fd < 0)
  257                  return 0; /* If we don't have a pidfd we cannot validate it, hence we assume it's all OK → return 0 */
  258

  260          if (r < 0)
  261                  return r;
  262
