import os, subprocess, sys, tempfile
sys.path.insert(0, sys.argv[1] + "/tools")
from architect_definitions import DEFS, ORDER, PROTOCOL, MODEL
REPO = "/home/nexus/NEXUS/nexus-os"; C = "d5964ee5326c2733177307db6f507328023d361a"
UNIT = "crates/nexus-governed-control/src/ledger/protocol/unit_tests.rs"
RF = os.path.expanduser("~/.rustup/toolchains/1.94.0-x86_64-unknown-linux-gnu/bin/rustfmt")
src = {p: subprocess.run(["git", "-C", REPO, "show", f"{C}:{p}"], capture_output=True, check=True).stdout.decode()
       for p in (PROTOCOL, MODEL, UNIT)}
only = sys.argv[2:] or ORDER
for cid in only:
    d = DEFS[cid]; base = src[d["file"]]; text = base; spans = []
    ok = True
    for e in d["edits"]:
        n = base.count(e["anchor"]); i = base.find(e["anchor"])
        if n != 1 or (i > 0 and base[i-1] != "\n"):
            print(cid, "ANCHOR COUNT", n, "linestart", i == 0 or base[i-1] == "\n"); ok = False
        spans.append((i, i + len(e["anchor"])))
    spans_s = sorted(spans)
    for a, b in zip(spans_s, spans_s[1:]):
        if a[1] > b[0]: print(cid, "OVERLAP"); ok = False
    for e in d["edits"]:
        assert text.count(e["anchor"]) == 1, cid
        text = text.replace(e["anchor"], e["replacement"], 1)
    with tempfile.TemporaryDirectory(dir=sys.argv[1] + "/dev-local") as t:
        for p in (PROTOCOL, MODEL, UNIT):
            q = os.path.join(t, "src", os.path.relpath(p, "crates/nexus-governed-control/src"))
            os.makedirs(os.path.dirname(q), exist_ok=True)
            open(q, "w").write(text if p == d["file"] else src[p])
        open(os.path.join(t, "empty.toml"), "w").close()
        tgt = os.path.join(t, "src", os.path.relpath(d["file"], "crates/nexus-governed-control/src"))
        r = subprocess.run([RF, "--check", "--edition", "2021", "--config-path", os.path.join(t, "empty.toml"), tgt],
                           capture_output=True, text=True, cwd=t)
        print(f"{cid}: anchors_ok={ok} rustfmt rc={r.returncode}")
        if r.returncode: print(r.stdout[-3000:], r.stderr[-2000:])
